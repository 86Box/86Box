cmake_minimum_required(VERSION 3.16)
project(QtSelectionTest LANGUAGES C CXX)

option(USE_QT6 "Require Qt6" OFF)
option(QT "Enable Qt" ON)
set(CMAKE_PREFIX_PATH "${TEST_PREFIX}/qt6;${TEST_PREFIX}/qt5")
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG ON)
set(_original_prefix "${CMAKE_PREFIX_PATH}")

if(QT)
    include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/FindQt.cmake")
    if(NOT QT_MAJOR EQUAL EXPECTED_MAJOR)
        message(FATAL_ERROR "Expected Qt${EXPECTED_MAJOR}, selected Qt${QT_MAJOR}")
    endif()
    if((QT_MAJOR EQUAL 6 AND NOT USE_QT6) OR (QT_MAJOR EQUAL 5 AND USE_QT6))
        message(FATAL_ERROR "USE_QT6 disagrees with QT_MAJOR")
    endif()
    get_property(_override CACHE USE_QT6 PROPERTY VALUE)
    if(NOT "${_override}" STREQUAL "${EXPECTED_OVERRIDE}")
        message(FATAL_ERROR "Detection changed the cached override to ${_override}")
    endif()
    if(NOT CMAKE_PREFIX_PATH STREQUAL _original_prefix OR NOT CMAKE_FIND_PACKAGE_PREFER_CONFIG)
        message(FATAL_ERROR "Detection leaked its package search settings")
    endif()
    if(MINGW AND QT_STATIC)
        if(NOT QT_TEST_PREFIX_${QT_MAJOR} STREQUAL "$ENV{MSYSTEM_PREFIX}/qt${QT_MAJOR}-static")
            message(FATAL_ERROR "Wrong MSYS2 static Qt search prefix")
        endif()
    elseif(NOT QT_TEST_PREFIX_${QT_MAJOR} STREQUAL _original_prefix)
        message(FATAL_ERROR "Wrong dynamic Qt search prefix")
    endif()
    if(VCPKG_TOOLCHAIN AND VCPKG_USE_HOST_TOOLS)
        if(NOT QT_HOST_PATH STREQUAL "${VCPKG_INSTALLED_DIR}/${VCPKG_HOST_TRIPLET}/tools/Qt${QT_MAJOR}")
            message(FATAL_ERROR "Wrong vcpkg host tools path: ${QT_HOST_PATH}")
        endif()
    endif()
    if(APPLE)
        if(QT_MAJOR EQUAL 6)
            set(_expected_deployment "10.15")
        elseif(MOLTENVK)
            set(_expected_deployment "10.14")
        else()
            set(_expected_deployment "10.13")
        endif()
        if(NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL _expected_deployment)
            message(FATAL_ERROR "Wrong macOS deployment target")
        endif()
    endif()
elseif(DEFINED QT_MAJOR)
    message(FATAL_ERROR "Qt was selected with QT=OFF")
endif()

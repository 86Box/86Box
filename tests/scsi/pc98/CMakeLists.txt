cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(pc98_scsi_tests C)
    enable_testing()
endif()
add_executable(pc98_scsi_tests ../pc98_55_test.c)
target_include_directories(pc98_scsi_tests PRIVATE ../../../src/include)
target_compile_features(pc98_scsi_tests PRIVATE c_std_11)
target_compile_definitions(pc98_scsi_tests PRIVATE "$<$<CONFIG:Release>:RELEASE_BUILD>")
target_compile_options(pc98_scsi_tests PRIVATE
    $<$<C_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra>)
add_test(NAME pc98_scsi COMMAND pc98_scsi_tests)
